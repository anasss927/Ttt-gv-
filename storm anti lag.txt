-- Anti-Lag GUI Script
local RunService = game:GetService('RunService')
local UserInputService = game:GetService('UserInputService')
local TweenService = game:GetService('TweenService')
local LocalPlayer = game:GetService('Players').LocalPlayer
local CoreGui = game:GetService('CoreGui')

-- Anti-lag state
local antiLagEnabled = false
local maxSpeedValue = 31
local antiLagConnection = nil
local targetSpeed = 31

-- Speed limiter function
local function setMaxSpeed(speed)
    targetSpeed = math.clamp(speed, 30, 32)
    maxSpeedValue = targetSpeed
    return targetSpeed
end

-- Anti-lag function
local function enableAntiLag()
    if antiLagConnection then antiLagConnection:Disconnect() end
    
    antiLagConnection = RunService.Heartbeat:Connect(function()
        if not antiLagEnabled then return end
        
        local character = LocalPlayer.Character
        if not character then return end
        
        local hrp = character:FindFirstChild('HumanoidRootPart')
        if not hrp then return end
        
        -- Disable proximity prompts in range
        for _, descendant in ipairs(workspace:GetDescendants()) do
            if descendant:IsA('ProximityPrompt') then
                local parent = descendant.Parent
                if parent:IsA('BasePart') and hrp then
                    local distance = (hrp.Position - parent.Position).Magnitude
                    if distance < descendant.MaxActivationDistance then
                        -- Prevent prompt activation
                        descendant.Enabled = false
                    else
                        descendant.Enabled = true
                    end
                end
            end
        end
        
        -- Limit player speed
        local velocity = hrp.AssemblyLinearVelocity
        local horizontalSpeed = math.sqrt((velocity.X * velocity.X) + (velocity.Z * velocity.Z))
        
        if horizontalSpeed > maxSpeedValue and UserInputService:IsKeyDown(Enum.KeyCode.W) then
            local scale = maxSpeedValue / horizontalSpeed
            hrp.AssemblyLinearVelocity = Vector3.new(velocity.X * scale, velocity.Y, velocity.Z * scale)
        end
    end)
end

local colorSeq = ColorSequence.new({
    ColorSequenceKeypoint.new(0, Color3.fromRGB(60, 65, 70)),
    ColorSequenceKeypoint.new(0.4, Color3.fromRGB(60, 65, 70)),
    ColorSequenceKeypoint.new(0.5, Color3.fromRGB(255, 255, 255)),
    ColorSequenceKeypoint.new(0.6, Color3.fromRGB(60, 65, 70)),
    ColorSequenceKeypoint.new(1, Color3.fromRGB(60, 65, 70)),
})
gradient.Color = colorSeq

-- Animate stroke
local strokeAnimation
strokeAnimation = RunService.Heartbeat:Connect(function(deltaTime)
    if not mainFrame.Parent then strokeAnimation:Disconnect() return end
    local time = tick()
    gradient.Rotation = (time * 40) % 360
    local sinVal = math.sin((time * 0.8 * math.pi) * 2)
    stroke.Transparency = 0.25 * (1 - ((sinVal + 1) / 2))
end)
