local function saveConfiguration()
    local configData = { autoBatKey = keybinds.autoBat.Name }
    pcall(function()
        writefile("EnvyAutoBatDesyncConfig.json", HttpService:JSONEncode(configData))
    end)
end

local function loadConfiguration()
    local success, hasFile = pcall(function() return isfile("EnvyAutoBatDesyncConfig.json") end)
    if not success or not hasFile then return end
    local successDecode, decodedConfig = pcall(function()
        return HttpService:JSONDecode(readfile("EnvyAutoBatDesyncConfig.json"))
    end)
    if successDecode and decodedConfig and decodedConfig.autoBatKey then
        local foundKey = Enum.KeyCode[decodedConfig.autoBatKey]
        if foundKey then
            keybinds.autoBat = foundKey
        end
    end
end

    isListening = false

    keybindButton.TextColor3 = WHITE
    if key then
        keybindButton.Text = key.Name
        keybinds.autoBat = key
        saveConfiguration()
    end
end

keybindButton.MouseButton1Click:Connect(function()
    if isListening then stopKeyListening(nil) return end
    isListening = true
    keybindButton.Text = "..."
    TweenService:Create(keyStroke, TweenInfo.new(0.1), {Color = WHITE}):Play()
    keyConnection = UserInputService.InputBegan:Connect(function(input)
        if not isListening then return end
        if input.UserInputType == Enum.UserInputType.Keyboard then
            if input.KeyCode == Enum.KeyCode.Escape then stopKeyListening(nil) return end
            stopKeyListening(input.KeyCode)
        end
    end)
end)

local function getBat()
    local character = LocalPlayer.Character
    if not character then return nil end
    local tool = character:FindFirstChild("Bat")
    if tool then return tool end
    local backpack = LocalPlayer:FindFirstChild("Backpack")
    if backpack then
        tool = backpack:FindFirstChild("Bat")
        if tool then tool.Parent = character return tool end
    end
    return nil
end

local function executeHit()
    if pluginState.isHitting then return end
    pluginState.isHitting = true
    pcall(function()
        local bat = getBat()
        if bat then
            bat:Activate()
            local remote = bat:FindFirstChildWhichIsA("RemoteEvent")
            if remote then remote:FireServer() end
        end
    end)
    task.delay(0.08, function() pluginState.isHitting = false end)
end

local function getClosestPlayer()
    if not localRootPart then return nil, math.huge end
    local closestPlayer, minDistance = nil, math.huge
    for _, player in pairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local root = player.Character:FindFirstChild("HumanoidRootPart")
            if root then
                local distance = (localRootPart.Position - root.Position).Magnitude
                if distance < minDistance then
                    minDistance = distance
                    closestPlayer = player
                end
            end
        end
    end
    return closestPlayer, minDistance
end

local function setupCharacter(character)
    task.wait(0.1)
    localHumanoid = character:WaitForChild("Humanoid", 5)
    localRootPart = character:WaitForChild("HumanoidRootPart", 5)
end

LocalPlayer.CharacterAdded:Connect(setupCharacter)
if LocalPlayer.Character then task.spawn(setupCharacter, LocalPlayer.Character) end

RunService.Heartbeat:Connect(function()
    if not (pluginState.isAutoBatEnabled and localHumanoid and localRootPart) then return end
    local target, distance = getClosestPlayer()
    if target and target.Character then
        local targetRoot = target.Character:FindFirstChild("HumanoidRootPart")
        if targetRoot then
            if sethiddenproperty then
                sethiddenproperty(localRootPart, "PhysicsRepRootPart", targetRoot)
            end
            local targetPosition = targetRoot.Position + Vector3.new(0, 0.9, 0)
            if (localRootPart.Position - targetPosition).Magnitude > 8 then
                localRootPart.CFrame = CFrame.new(targetPosition)
            end
            local camera = workspace.CurrentCamera
            camera.CFrame = CFrame.new(camera.CFrame.Position, targetRoot.Position)
            executeHit()
        end
    end
end)

UserInputService.InputBegan:Connect(function(input, isGameProcessed)
    if isGameProcessed then return end
    if input.UserInputType == Enum.UserInputType.Keyboard then
        if input.KeyCode == keybinds.autoBat then
            pluginState.isAutoBatEnabled = not pluginState.isAutoBatEnabled
            updateToggleVisual(pluginState.isAutoBatEnabled)
        elseif input.KeyCode == keybinds.toggleGui then
            pluginState.isGuiVisible = not pluginState.isGuiVisible
            mainFrame.Visible = pluginState.isGuiVisible
        end
    end
end)
