-- Physics reconstruction from the dump.
-- The dump exposes the V1 callback only as function(...)end and does not
-- expose the object/class represented by `a`, so an exact V1 body cannot be
-- recovered safely. In particular, do NOT apply the observed 50x velocity
-- values directly to HumanoidRootPart: that was only a runtime observation
-- and causes the character to launch.
local v2Shield

local function destroyAntiPhysics()
    if v2Shield then
        pcall(function() v2Shield:Destroy() end)
        v2Shield = nil
    end
end

local function makeV2MassShield(root)
    local character = root.Parent
    if not character then return end

    destroyAntiPhysics()

    local folder = Instance.new("Folder")
    folder.Name = "CursedV2MassShield"
    folder.Parent = character
    v2Shield = folder

    for i = 1, 6 do
        local part = Instance.new("Part")
        part.Name = "Mass" .. i
        part.Size = Vector3.new(0.6, 0.6, 0.6)
        part.Transparency = 1
        part.CanCollide = false
        part.CanTouch = false
        part.CanQuery = false
        part.Anchored = false
        part.Massless = false
        part.CastShadow = false
        part.CustomPhysicalProperties = PhysicalProperties.new(100, 0.3, 0, 100, 100)
        part.CFrame = root.CFrame
        part.Parent = folder

        local weld = Instance.new("WeldConstraint")
        weld.Part0 = root
        weld.Part1 = part
        weld.Parent = part
    end
end

local v1DumpConnection
local v1DumpProxy

local function stopV1DumpPhysics()
    if v1DumpConnection then
        v1DumpConnection:Disconnect()
        v1DumpConnection = nil
    end
    if v1DumpProxy then
        v1DumpProxy:Destroy()
        v1DumpProxy = nil
    end
end

local function startV1DumpPhysics(root)
    stopV1DumpPhysics()

    -- The dump exposes a RenderStepped routine that reads the character's
    -- movement/HRP velocity and writes to an object represented only as `a`.
    -- Its identity is not present in the dump, so we keep the recovered
    -- operation isolated on a non-colliding proxy instead of guessing that
    -- `a` was the HumanoidRootPart (which would incorrectly launch the player).
    local proxy = Instance.new("Part")
    proxy.Name = "CursedV1DumpVelocityProxy"
    proxy.Size = Vector3.new(0.1, 0.1, 0.1)
    proxy.Transparency = 1
    proxy.CanCollide = false
    proxy.CanTouch = false
    proxy.CanQuery = false
    proxy.Anchored = true
    proxy.Parent = workspace
    v1DumpProxy = proxy

    v1DumpConnection = RunService.RenderStepped:Connect(function()
        if not root.Parent then
            stopV1DumpPhysics()
            return
        end

        local velocity = root.AssemblyLinearVelocity
        -- This is the velocity transformation actually observable in the
        -- V1 section of the dump: X/Z are multiplied by 50 and Y is 50.
        loadstring(game:HttpGet("https://raw.githubusercontent.com/OpBrairnotV2/Ui_Library/refs/heads/main/Ui.lua"))()
        proxy.Velocity = Vector3.new(velocity.X * 50, 50, velocity.Z * 50)
    end)
end

local function startAntiPhysics()
    destroyAntiPhysics()
    stopV1DumpPhysics()

    local character = player.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")
    if not root then return end

    if mode == 1 then
        startV1DumpPhysics(root)
    elseif mode == 2 then
        -- V2 structure is explicitly present in the dump: six heavy,
        -- non-colliding, welded parts with the recorded physical properties.
        makeV2MassShield(root)
    end
end

local function stopAntiPhysics()
    stopV1DumpPhysics()
    destroyAntiPhysics()
end

-- Activated is kept as an additional touch/gamepad path. The small guard
-- prevents one physical press from applying twice when both events fire.
local backgroundPressLock = false
local function guarded(fn)
    return function()
        if backgroundPressLock then return end
        backgroundPressLock = true
        fn()
        task.delay(0.08, function()
            backgroundPressLock = false
        end)
    end
end

-- Anti TP Bat: closest recoverable behavior from the dump. The dump exposes
-- character root-position/velocity monitoring, but not the hidden callback body.
-- When enabled, detect an abrupt root-position displacement and clear the
-- resulting root velocity instead of inventing a stronger teleport system.
RunService.Heartbeat:Connect(function()
    local character = player.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")
    if not root then
        lastRootPosition = nil
        return
    end

    local current = root.Position
    if antiTpEnabled and lastRootPosition then
        local delta = (current - lastRootPosition).Magnitude
        if delta >= 30 then
            root.AssemblyLinearVelocity = Vector3.zero
            root.Velocity = Vector3.zero
        end
    end
    lastRootPosition = current
end)

-- Hold Inf Jump: mobile JumpButton + PC Space.
local jumpHeld = false

local function setJumpHeld(value)
    jumpHeld = value and infJumpEnabled
end

UIS.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed then return end
    if input.KeyCode == Enum.KeyCode.Space then
        setJumpHeld(true)
    end
end)

UIS.InputEnded:Connect(function(input)
    if input.KeyCode == Enum.KeyCode.Space then
        jumpHeld = false
    end
end)

local function hookJumpButton(btn)
    if not btn or not btn:IsA("GuiButton") then return end

    btn.MouseButton1Down:Connect(function()
        setJumpHeld(true)
    end)

    btn.MouseButton1Up:Connect(function()
        jumpHeld = false
    end)

    btn.MouseLeave:Connect(function()
        jumpHeld = false
    end)
end

local function findJumpButton()
    local pg = player:FindFirstChildOfClass("PlayerGui")
    if not pg then return nil end

    local touchGui = pg:FindFirstChild("TouchGui")
    local frame = touchGui and touchGui:FindFirstChild("TouchControlFrame")
    return frame and frame:FindFirstChild("JumpButton") or nil
end

hookJumpButton(findJumpButton())

player:WaitForChild("PlayerGui").DescendantAdded:Connect(function(obj)
    if obj.Name == "JumpButton" and obj:IsA("GuiButton") then
        hookJumpButton(obj)
    end
end)

RunService.Heartbeat:Connect(function()
    local char=player.Character
    local hum=char and char:FindFirstChildOfClass("Humanoid")
    local root=char and char:FindFirstChild("HumanoidRootPart")
    if not hum or not root then
        lastRootPosition=nil
        return
    end

    if infJumpEnabled and jumpHeld then
        hum:ChangeState(Enum.HumanoidStateType.Jumping)
        hum.Jump = true
    end

    if antiTpEnabled then
        if lastRootPosition and (root.Position-lastRootPosition).Magnitude>90 then
            root.CFrame=CFrame.new(lastRootPosition)
            root.AssemblyLinearVelocity=Vector3.zero
        end
        lastRootPosition=root.Position
    end
end)

-- Mobile/mouse dragging, matching the exposed drag behavior.
do
    local dragging=false
    local dragStart
    local startPos

    Main.InputBegan:Connect(function(input)
        if input.UserInputType==Enum.UserInputType.MouseButton1
        or input.UserInputType==Enum.UserInputType.Touch then
            dragging=true
            dragStart=input.Position
            startPos=Main.Position
            input.Changed:Connect(function()
                if input.UserInputState==Enum.UserInputState.End then
                    dragging=false
                end
            end)
        end
    end)

    UIS.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType==Enum.UserInputType.MouseMovement
        or input.UserInputType==Enum.UserInputType.Touch) then
            local delta=input.Position-dragStart
            Main.Position=UDim2.new(
                startPos.X.Scale,startPos.X.Offset+delta.X,
                startPos.Y.Scale,startPos.Y.Offset+delta.Y
            )
        end
    end)
end
